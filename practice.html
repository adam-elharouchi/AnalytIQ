# Function to randomly set a fraction of values to NaN in a DataFrame
def introduce_missing_values(df, fraction_missing, random_state=None):
    np.random.seed(random_state)
    mask = np.random.rand(*df.shape) < fraction_missing
    df_with_missing = df.mask(mask)
    return df_with_missing

# Define a function to count outliers in a column
def count_outliers(column):
    if pd.api.types.is_numeric_dtype(column):
        Q1 = column.quantile(0.25)
        Q3 = column.quantile(0.75)
        IQR = Q3 - Q1

        # Calculate the lower and upper bounds for outliers
        lower_bound = Q1 - 1.5 * IQR
        upper_bound = Q3 + 1.5 * IQR

        # Count outliers
        outliers = column[(column < lower_bound) | (column > upper_bound)]
        return outliers.shape[0]
    else:
        return 0  # For non-numeric columns, consider 0 outliers

# Filter numeric columns
numeric_columns = california_df.select_dtypes(include=['number'])

# Count outliers in each numeric column
outliers_count = numeric_columns.apply(count_outliers)

# Display the number of outliers in each numeric column
print("Number of outliers in each numeric column:")
print(outliers_count)

# Define a function to remove outliers in a column
def remove_outliers(column):
    # Calculate IQR
    Q1 = column.quantile(0.25)
    Q3 = column.quantile(0.75)
    IQR = Q3 - Q1

    # Calculate the lower and upper bounds for outliers
    lower_bound = Q1 - 1.5 * IQR
    upper_bound = Q3 + 1.5 * IQR

    # Remove outliers
    column = column[(column >= lower_bound) & (column <= upper_bound)]

    return column

# Create a copy of the DataFrame to avoid modifying the original data
california_df_clean = california_df.copy()

# Remove outliers from each numeric column
for column in california_df_clean.select_dtypes(include=['number']):
    california_df_clean[column] = remove_outliers(california_df_clean[column])

# Drop rows with NaN values
california_df_clean = california_df_clean.dropna()

# Display the cleaned DataFrame
print("

print(california_df_clean)
